main <- df |>
  select(id, matches("^(boredom|devices)_\\d+€_\\d+$")) |>
  pivot_longer(
    cols = -id,
    names_to = c("task", "amount", "level"),
    names_sep = "_",
    values_to = "response"

extras <- df |>
  select(id, ends_with("_pleasance"), ends_with("_howlong")) |>

  )

df_long <- main |>

df_long <- df_long |>

Hypothesis 1: More people will pay to leave under the “boredom” condition

Each person has 6 scenarios following the “boredom” condition and 6 scenarios following the “devices” condition. Their aggregated “score” is the yes-rate out of the 6 for each of the “boredom” and “devices” condition.

The null hypothesis is that people will pay the same under both conditions, or less under the “boredom” condition. The alternative hypothesis is that more people will pay to leave under the “boredom” condition.

We do paired t-tests throughout this report because samples are linked by participant in our within-subject design, and we assume participants are choosing independently, and these differences are normally distributed.

# Hypothesis 1
per_task <- df_long |>
  group_by(id, task) |>
  summarise(prop_yes = mean(response, na.rm = TRUE), .groups = "drop") |>
  pivot_wider(names_from = task, values_from = prop_yes)

t.test(per_task$boredom, per_task$devices, paired = TRUE, alternative = "greater")    # greater: test that boredom > devices

Hypothesis 2: The higher cost to leave, the less people will pay to leave

Two alternative ways of testing this: only compare cost 1 vs cost 9, or calculate slope of regression of proportion of “yes” on cost (this way, all three cost levels are factored in). We do one-sided t-tests, expecting fewer people to choose “yes” to leave as costs increase.

# Hypothesis 2
per_level <- df_long |>
  group_by(id, level) |>
  summarise(prop_yes = mean(response, na.rm = TRUE), .groups = "drop")

# (a) only compares level 1 vs level 9
lv <- per_level |>
  filter(level %in% c(1, 9)) |>
  pivot_wider(names_from = level, values_from = prop_yes, names_prefix = "level")

t.test(lv$level9, lv$level1, paired = TRUE, alternative = "less")   # expects fewer "yes to leave" at cost 9

Both tests turn out to yield significant results with p-values < 0.05. For the second option, mean slope is -0.048 and 95% CI is (-Inf, -0.03). This is expected, negative slope means that increase in cost is correlated with reduction in proportion of “yes”.

Hypothesis 3: The higher the budget, the more people pay to leave

This hypothesis is because we expect that if people have more means, they would be able to afford a more expensive option to save more of their time. This is a similar idea as 1, but we now calculate proportions out of 6 conditions for 10 euros vs. 20 euros budget.

per_amount <- df_long |>

cell_df <- df_long |>
  mutate(level = as.numeric(as.character(level))) |>
  group_by(task, amount, level) |>
  summarise(
    p  = mean(response, na.rm = TRUE),
    n  = sum(!is.na(response)),
    se = sqrt(p * (1 - p) / n),
    .groups = "drop"
  )

amount_labels <- c("10" = "Budget: 10€",
                   "20" = "Budget: 20€")

all_plot <- ggplot(cell_df, aes(x = level, y = p, colour = task)) +
  geom_line() +
  geom_point(size = 3) +
  geom_errorbar(aes(ymin = p - se, ymax = p + se), width = 0.3) +
  facet_wrap(~ amount, labeller = as_labeller(amount_labels)) +
  scale_x_continuous(breaks = c(1, 5, 9)) +

    limits = c(0, 1),
